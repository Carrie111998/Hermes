cd C:\Users\ISLAM\Desktop\hermesagent
$env:HERMES_HOME="$PWD\server\.test-hermes"

@'

  webui_enabled: true

  max_upload_bytes: 26214400

.\.venv\Scripts\python.exe -m server seed-demo `
